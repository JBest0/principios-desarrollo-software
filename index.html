public double calcularTotal(Pedido pedido) {
    double total = 0;
    for (Item item : pedido.getItems()) {
        total += item.getPrecio() * item.getCantidad();
    }
    if (pedido.getTipoCliente().equals("VIP")) {
        total *= 0.9;
    } else if (pedido.getTipoCliente().equals("MAYORISTA")) {
        total *= 0.85;
    }
    return total;
}
